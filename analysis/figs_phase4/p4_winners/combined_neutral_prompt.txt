        import numpy as np

class OM_CMA_EIS:
    def __init__(self, budget, dim):
        self.budget = budget
        self.dim = dim
        self.lower_bound = -5.0
        self.upper_bound = 5.0

        # Population sizing optimized for noise-resilient gradient estimation
        self.pop_size = 4 + int(3 * np.log(self.dim))
        if self.pop_size % 2 != 0:
            self.pop_size += 1

        self.mu = self.pop_size // 2
        self.weights = np.log(self.mu + 0.5) - np.log(np.arange(1, self.mu + 1))
        self.weights /= np.sum(self.weights)
        self.mueff = (np.sum(self.weights)**2) / np.sum(self.weights**2)

        # Strategy Parameters
        self.xmean = np.random.uniform(-1.0, 1.0, self.dim)
        self.sigma = 0.3 * (self.upper_bound - self.lower_bound)

        # Evolution Paths and Matrices
        self.pc = np.zeros(self.dim)
        self.ps = np.zeros(self.dim)
        self.C = np.eye(self.dim)
        self.chiN = np.sqrt(self.dim) * (1 - 1 / (4 * self.dim) + 1 / (21 * self.dim**2))

        # Learning rates
        self.cc = (4 + self.mueff / self.dim) / (self.dim + 4 + 2 * self.mueff / self.dim)
        self.cs = (self.mueff + 2) / (self.dim + self.mueff + 5)
        self.c1 = 2 / ((self.dim + 1.3)**2 + self.mueff)
        self.cmu = min(1 - self.c1, 2 * (self.mueff - 2 + 1/self.mueff) / ((self.dim + 2)**2 + self.mueff))
        self.damps = 1 + 2 * max(0, np.sqrt((self.mueff - 1) / (self.dim + 1)) - 1) + self.cs

        # Edge-Intensification Memory
        self.edge_momentum = np.zeros(self.dim)
        self.f_opt = np.inf
        self.x_opt = None

    def __call__(self, func):
        evals = 0
        generation = 0

        while evals < self.budget:
            try:
                # Spectral decomposition for coordinate transformation
                D_vals, B = np.linalg.eigh(self.C)
                if np.any(D_vals <= 0):
                    D_vals = np.maximum(D_vals, 1e-14)
                D = np.sqrt(D_vals)
            except (np.linalg.LinAlgError, ValueError):
                self.C = np.eye(self.dim)
                D, B = np.ones(self.dim), np.eye(self.dim)

            # 1. Orthogonal Mirror Sampling (EIS)
            # This maximizes the spread (edge density) and reduces plateau probability
            half_pop = self.pop_size // 2
            z_half = np.random.randn(half_pop, self.dim)
            if self.dim >= half_pop:
                q, _ = np.linalg.qr(z_half.T)
                z_half = q.T * np.sqrt(self.dim)

            # Create mirrored population
            pop_z = np.vstack([z_half, -z_half])
            pop_x = np.zeros((self.pop_size, self.dim))
            fitness = np.zeros(self.pop_size)

            # Apply transformation and Edge Momentum
            for i in range(self.pop_size):
                if evals >= self.budget:
                    fitness[i:] = np.inf
                    break

                # Project transform
                raw_step = B @ (D * pop_z[i])

                # Inject Edge Momentum: Bias search along the most productive recent gradient
                if generation > 2:
                    intensification_bias = 0.15 * self.edge_momentum * np.linalg.norm(raw_step)
                    step = raw_step + intensification_bias
                else:
                    step = raw_step

                pop_x[i] = self.xmean + self.sigma * step

                # Boundary handling using reflection to keep points near boundaries (High Edge Density)
                for d in range(self.dim):
                    if pop_x[i, d] < self.lower_bound:
                        pop_x[i, d] = self.lower_bound + (self.lower_bound - pop_x[i, d]) % (self.upper_bound - self.lower_bound)
                    elif pop_x[i, d] > self.upper_bound:
                        pop_x[i, d] = self.upper_bound - (pop_x[i, d] - self.upper_bound) % (self.upper_bound - self.lower_bound)
                    pop_x[i, d] = np.clip(pop_x[i, d], self.lower_bound, self.upper_bound)

                fitness[i] = func(pop_x[i])
                evals += 1

                if fitness[i] < self.f_opt:
                    self.f_opt = fitness[i]
                    self.x_opt = pop_x[i].copy()

            if evals >= self.budget: break

            # 2. Ranking and Statistics Update
            indices = np.argsort(fitness)
            best_indices = indices[:self.mu]

            x_old = self.xmean.copy()
            self.xmean = np.dot(self.weights, pop_x[best_indices])

            # Update paths
            y_mean = (self.xmean - x_old) / self.sigma
            inv_sqrt_C = B @ np.diag(1.0 / D) @ B.T
            self.ps = (1 - self.cs) * self.ps + np.sqrt(self.cs * (2 - self.cs) * self.mueff) * (inv_sqrt_C @ y_mean)

            ps_norm = np.linalg.norm(self.ps)
            hsig = ps_norm / np.sqrt(1 - (1 - self.cs)**(2 * (evals / self.pop_size))) / self.chiN < 1.4 + 2 / (self.dim + 1)
            self.pc = (1 - self.cc) * self.pc + hsig * np.sqrt(self.cc * (2 - self.cc) * self.mueff) * y_mean

            # 3. Edge Momentum Update (Gradient Memory)
            # Tracks normalized direction of success to push through plateaus
            if np.linalg.norm(y_mean) > 1e-15:
                current_grad = y_mean / np.linalg.norm(y_mean)
                self.edge_momentum = (1 - self.cc) * self.edge_momentum + self.cc * current_grad

            # 4. CMA Matrix and Step Size Adaptation
            artmp = (pop_x[best_indices] - x_old) / self.sigma
            rank_mu = np.dot(artmp.T, (self.weights[:, None] * artmp))
            rank_1 = np.outer(self.pc, self.pc)

            self.C = (1 - self.c1 - self.cmu) * self.C \
                     + self.c1 * (rank_1 + (1 - hsig) * self.cc * (2 - self.cc) * self.C) \
                     + self.cmu * rank_mu

            # PSA (Path Step-size Adaptation)
            self.sigma *= np.exp((self.cs / self.damps) * (ps_norm / self.chiN - 1))

            # Plateaus detection: if fitness diversity is too low, boost exploration
            if abs(fitness[indices[0]] - fitness[indices[-1]]) < 1e-12:
                self.sigma *= 1.1

            generation += 1

        return self.f_opt, self.x_opt
