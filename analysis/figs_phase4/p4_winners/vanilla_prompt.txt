class IGA_CMA:
    def __init__(self, budget, dim):
        self.budget = budget
        self.dim = dim
        self.lb = -5.0
        self.ub = 5.0
        self.evals = 0

    def __call__(self, func):
        f_best = np.inf
        x_best = None
        self.evals = 0
        n = self.dim

        # IPOP strategy: increase population size on restart
        pop_multiplier = 1.0

            lambda_ = int((4 + np.floor(3 * np.log(n))) * pop_multiplier)
            mu = lambda_ // 2
            weights = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))
            weights /= np.sum(weights)
            mueff = (np.sum(weights)**2) / np.sum(weights**2)

            # Adaptation constants
            cc = (4 + mueff / n) / (n + 4 + 2 * mueff / n)
            cs = (mueff + 2) / (n + mueff + 5)
            c1 = 2 / ((n + 1.3)**2 + mueff)
            cmu = min(1 - c1, 2 * (mueff - 2 + 1 / mueff) / ((n + 2)**2 + mueff))
            damps = 1 + 2 * max(0, np.sqrt((mueff - 1) / (n + 1)) - 1) + cs
            chiN = n**0.5 * (1 - 1 / (4 * n) + 1 / (21 * n**2))

            # 2. Initialization
            if x_best is not None and np.random.rand() < 0.3:
                x_mean = x_best + 0.1 * np.random.randn(n)
            else:
                x_mean = np.random.uniform(self.lb * 0.6, self.ub * 0.6, n)

            sigma = 0.25 * (self.ub - self.lb)
            pc, ps = np.zeros(n), np.zeros(n)
            B, D = np.eye(n), np.ones(n)
            C = np.eye(n)

            best_f_run = np.inf
            history_f = []

                # 3. Mirrored Orthogonal Sampling
                z_half = np.random.randn(n, lambda_ // 2)
                if n > 1:

                    z_half = q @ np.diag(np.sign(np.diag(r))) * np.sqrt(n)

                Z = np.hstack([z_half, -z_half])
                if Z.shape[1] < lambda_:
                    Z = np.hstack([Z, np.random.randn(n, lambda_ - Z.shape[1])])

                BD = B * D
                X = x_mean[:, None] + sigma * (BD @ Z)
                X = np.clip(X.T, self.lb, self.ub)

                # 4. Evaluation
                arfitness = np.zeros(lambda_)
                for k in range(lambda_):
                    if self.evals >= self.budget:
                        arfitness[k:] = np.inf
                        break
                    arfitness[k] = func(X[k])
                    self.evals += 1
                    if arfitness[k] < f_best:
                        f_best = arfitness[k]
                        x_best = X[k].copy()

                if self.evals >= self.budget: break

                # 5. Sorting and Selection
                idx = np.argsort(arfitness)
                best_f_curr = arfitness[idx[0]]

                if best_f_curr < best_f_run - 1e-14:
                    best_f_run = best_f_curr
                    stagnation_count = 0
                else:
                    stagnation_count += 1

                x_old = x_mean.copy()
                x_mean = X[idx[:mu]].T @ weights

                # 6. CMA Path and Covariance Updates (Active CMA)
                inv_sqrt_C = B @ np.diag(1.0/D) @ B.T
                ps = (1 - cs) * ps + np.sqrt(cs * (2 - cs) * mueff) * inv_sqrt_C @ (x_mean - x_old) / sigma
                hsig = np.linalg.norm(ps) / np.sqrt(1 - (1 - cs)**(2 * self.evals / lambda_)) / chiN < 1.4 + 2 / (n + 1)
                pc = (1 - cc) * pc + hsig * np.sqrt(cc * (2 - cc) * mueff) * (x_mean - x_old) / sigma

                y = (X[idx[:mu]] - x_old) / sigma
                C_pos = y.T @ np.diag(weights) @ y

                neg_weights = weights[::-1]
                y_neg = (X[idx[-mu:]] - x_old) / sigma
                C_neg = y_neg.T @ np.diag(neg_weights) @ y_neg

                C = (1 - c1 - cmu) * C + \
                    c1 * (np.outer(pc, pc) + (1 - hsig) * cc * (2 - cc) * C) + \
                    cmu * (C_pos - 0.8 * C_neg) # Aggressive active update

                sigma *= np.exp((cs / damps) * (np.linalg.norm(ps) / chiN - 1))

                # Decomposition
                C = np.triu(C) + np.triu(C, 1).T
                try:
                    vals, B = np.linalg.eigh(C)
                    vals = np.maximum(vals, 1e-15)
                    D = np.sqrt(vals)
                except: break

                # 7. DE-Injection Step (Hybridization)
                if stagnation_count > 5 and self.evals < self.budget - 3:
                    # Differential mutation using top performers
                    r1, r2, r3 = idx[0], idx[1], idx[2]
                    v_de = X[r1] + 0.5 * (X[r2] - X[r3])
                    v_de = np.clip(v_de, self.lb, self.ub)
                    f_de = func(v_de)
                    self.evals += 1
                    if f_de < f_best:
                        f_best = f_de
                        x_best = v_de.copy()
                        x_mean = v_de # Shift mean towards DE success

                # 8. Convergence/Restart Logic
                history_f.append(best_f_curr)
                if len(history_f) > 20: history_f.pop(0)

                if sigma < 1e-12 or (len(history_f) == 20 and np.std(history_f) < 1e-15):
                    pop_multiplier *= 2
                    break
                if stagnation_count > 30 + 5*n:
                    pop_multiplier *= 1.5
                    break

        return f_best, x_best
